cmake_minimum_required(VERSION 3.16)
project(myroboview_backend)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rosidl_typesupport_introspection_cpp REQUIRED)
find_package(Drogon CONFIG REQUIRED)
add_library(backend_core src/config.cpp src/state.cpp src/introspection.cpp src/navigation.cpp)
target_include_directories(backend_core PUBLIC include)
ament_target_dependencies(backend_core rclcpp ament_index_cpp rosidl_typesupport_introspection_cpp)
target_link_libraries(backend_core Drogon::Drogon)
add_executable(myroboview_backend_node src/bridge.cpp src/api.cpp src/websocket.cpp)
ament_target_dependencies(myroboview_backend_node rclcpp ament_index_cpp)
target_link_libraries(myroboview_backend_node backend_core)
foreach(target backend_core myroboview_backend_node)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
if(BUILD_TESTING)
  enable_testing()
  find_package(node_app_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  add_executable(core_test test/core_test.cpp)
  ament_target_dependencies(core_test node_app_msgs sensor_msgs)
  target_link_libraries(core_test backend_core)
  add_test(NAME core_test COMMAND core_test ${CMAKE_CURRENT_SOURCE_DIR}/config/myroboview.json ${CMAKE_CURRENT_SOURCE_DIR}/assets/navigation.json)
endif()
install(TARGETS myroboview_backend_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config web assets DESTINATION share/${PROJECT_NAME})
ament_package()
