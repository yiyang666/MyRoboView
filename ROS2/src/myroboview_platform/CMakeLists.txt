cmake_minimum_required(VERSION 3.16)
project(myroboview_platform)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rosidl_typesupport_introspection_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(myroboview_interfaces REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
find_package(jsoncpp CONFIG REQUIRED)
add_library(platform_core src/config.cpp src/state.cpp src/introspection.cpp)
target_include_directories(platform_core PUBLIC include)
ament_target_dependencies(platform_core rclcpp ament_index_cpp rosidl_typesupport_introspection_cpp)
target_link_libraries(platform_core jsoncpp_lib)
add_executable(bridge src/bridge.cpp src/http.cpp)
ament_target_dependencies(bridge rclcpp ament_index_cpp)
target_link_libraries(bridge platform_core Boost::system)
add_executable(mock src/mock.cpp)
ament_target_dependencies(mock rclcpp sensor_msgs nav_msgs myroboview_interfaces)
target_link_libraries(mock platform_core)
foreach(target platform_core bridge mock)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test test/core_test.cpp)
  ament_target_dependencies(core_test sensor_msgs nav_msgs myroboview_interfaces)
  target_link_libraries(core_test platform_core)
  add_test(NAME core_test COMMAND core_test ${CMAKE_CURRENT_SOURCE_DIR}/config/demo.json)
endif()
install(TARGETS bridge mock DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config web DESTINATION share/${PROJECT_NAME})
ament_package()
